import React, { useState, useEffect, useRef } from 'react';

  <svg className="w-5 h-5 text-red-400" fill="none" stroke="currentColor" viewBox="0 0 24 24">

  <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">

);

export default function App() {
  const [apiKey, setApiKey] = useState('');
  const [apiInputVisible, setApiInputVisible] = useState(false);
  const [imageSrc, setImageSrc] = useState(null);
  const [imageFile, setImageFile] = useState(null);
  const [targetName, setTargetName] = useState('Ibran Khan');
  const [headline, setHeadline] = useState('Happy Birthday');
  const [theme, setTheme] = useState('gold-black'); // 'gold-black', 'watercolor-ink', 'cinematic-neon', 'minimalist-white'
  const [lighting, setLighting] = useState('cinematic-golden');
  const [composition, setComposition] = useState('double-portrait'); // 'double-portrait', 'flying-hair', 'poster-collage'
  const [extraPrompt, setExtraPrompt] = useState('Ultra-realistic birthday collage poster, stylish Indian look, double portrait composition, one large faded background with ink splash effect, soft cinematic lighting, elegant typography, aesthetic studio photo manipulation, smooth skin, trendy fashion, 16k resolution.');
  
  const [isGenerating, setIsGenerating] = useState(false);
  const [generationSteps, setGenerationSteps] = useState('');
  const [generatedImageUrl, setGeneratedImageUrl] = useState(null);
  const [errorMessage, setErrorMessage] = useState('');
  const [recentPosters, setRecentPosters] = useState([]);

  // LLM Added Features State
  const [isEnhancingPrompt, setIsEnhancingPrompt] = useState(false);
  const [isGeneratingWishes, setIsGeneratingWishes] = useState(false);
  const [aiGeneratedWishes, setAiGeneratedWishes] = useState([]);
  const [copySuccessId, setCopySuccessId] = useState(null);
  const [llmNotification, setLlmNotification] = useState('');

  const fileInputRef = useRef(null);

  // Load API key from local storage if available
  useEffect(() => {
    const savedKey = localStorage.getItem('gemini_poster_api_key');
    if (savedKey) {
      setApiKey(savedKey);
    } else {
      setApiInputVisible(true);
    }

    // Load custom history
    const savedHistory = localStorage.getItem('gemini_poster_history');
    if (savedHistory) {
      try {
        setRecentPosters(JSON.parse(savedHistory));
      } catch (e) {
        console.error(e);
      }
    }
  }, []);

  const handleApiKeyChange = (e) => {
    const key = e.target.value;
    setApiKey(key);
    localStorage.setItem('gemini_poster_api_key', key);
  };

  const handleImageUpload = (e) => {
    const file = e.target.files[0];
    if (file) {
      const reader = new FileReader();
      reader.onload = (event) => {
        setImageSrc(event.target.result);
        setImageFile(file);
      };
      reader.readAsDataURL(file);
    }
  };

  const triggerFileUpload = () => {
    fileInputRef.current?.click();
  };

  const removeUploadedImage = () => {
    setImageSrc(null);
    setImageFile(null);
    if (fileInputRef.current) {
      fileInputRef.current.value = '';
    }
  };

  // Helper function for API calls with Exponential Backoff
  const fetchWithRetry = async (url, options, maxRetries = 5) => {
    let delay = 1000;
    for (let attempt = 0; attempt < maxRetries; attempt++) {
      try {
        const response = await fetch(url, options);
        if (response.ok) {
          return await response.json();
        }
        
        // If server error or rate limit, retry
        if (response.status === 429 || response.status >= 500) {
          await new Promise((resolve) => setTimeout(resolve, delay));
          delay *= 2; // exponential backoff
          continue;
        }
        
        const errText = await response.text();
        throw new Error(`API Error: ${response.status} - ${errText}`);
      } catch (error) {
        if (attempt === maxRetries - 1) {
          throw error;
        }
        await new Promise((resolve) => setTimeout(resolve, delay));
        delay *= 2;
      }
    }
  };

  // Safe helper to copy text to clipboard inside iframe environment
  const copyTextToClipboard = (text, id) => {
    const textArea = document.createElement("textarea");
    textArea.value = text;
    textArea.style.position = "fixed";  // Avoid scrolling to bottom
    document.body.appendChild(textArea);
    textArea.focus();
    textArea.select();
    try {
      document.execCommand('copy');
      setCopySuccessId(id);
      setTimeout(() => setCopySuccessId(null), 2000);
    } catch (err) {
      console.error('Fallback copy method failed', err);
    }
    document.body.removeChild(textArea);
  };

  // ✨ INTEGRATION 1: LLM Prompt Enhancer via gemini-2.5-flash-preview-09-2025
  const handleEnhancePrompt = async () => {
    if (!apiKey) {
      setErrorMessage('Please configure your Gemini API key first.');
      setApiInputVisible(true);
      return;
    }
    setIsEnhancingPrompt(true);
    setLlmNotification('');
    try {
      const systemInstruction = "You are an elite, professional graphic designer and creative director specialized in high-end birthday poster prompt writing.";
      const promptQuery = `Take the following poster configurations and expand them into a masterpiece description.
Celebrant Name: ${targetName}
Occasion: ${headline}
Layout Composition: ${composition}
Theme Colors: ${theme}
Lighting Style: ${lighting}
Basic details: ${extraPrompt}

Generate a single paragraph prompt that includes beautiful aesthetics, 16k details, smooth skin lighting, graphic design layout details, elegant typography positioning. Do NOT include greeting text or intro remarks. Output ONLY the raw expanded prompt text.`;

      const payload = {
        contents: [{ parts: [{ text: promptQuery }] }],
        systemInstruction: { parts: [{ text: systemInstruction }] }
      };

      const targetUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
      const response = await fetchWithRetry(targetUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      const text = response.candidates?.[0]?.content?.parts?.[0]?.text;
      if (text) {
        setExtraPrompt(text.trim());
        setLlmNotification('✨ Prompt optimized with high-end designer instructions!');
        setTimeout(() => setLlmNotification(''), 4000);
      } else {
        throw new Error('Invalid response received from LLM Model.');
      }
    } catch (err) {
      console.error(err);
      setErrorMessage(`Prompt enhancement failed: ${err.message}`);
    } finally {
      setIsEnhancingPrompt(false);
    }
  };

  // ✨ INTEGRATION 2: LLM Social Caption & Shayari Composer via gemini-2.5-flash-preview-09-2025
  const handleComposeWishes = async () => {
    if (!apiKey) {
      setErrorMessage('Please configure your Gemini API key first.');
      setApiInputVisible(true);
      return;
    }
    setIsGeneratingWishes(true);
    try {
      const promptQuery = `Compose 3 distinct and absolutely beautiful birthday wishes, Instagram captions, or emotional status lines for ${targetName}'s birthday. 
The poster theme is "${theme}" with "${lighting}" lighting. Customize the tone of the wishes to match the mood:
- Style 1: Touchy, poetic, emotional shayari/lines (using English letters, like Urdu/Hindi in Roman script or pure English with grand impact).
- Style 2: High-fashion, aesthetic, trendy short Instagram caption with hashtags.
- Style 3: Royal, majestic, VIP greeting.

Return the result strictly as a JSON object containing an array of 3 objects, each having:
- "type": the tone category
- "text": the composed wish message with appropriate emojis
- "hashtags": a string of relevant trendy hashtags.

Use JSON schema structure.`;

      const payload = {
        contents: [{ parts: [{ text: promptQuery }] }],
        generationConfig: {
          responseMimeType: "application/json",
          responseSchema: {
            type: "OBJECT",
            properties: {
              wishes: {
                type: "ARRAY",
                items: {
                  type: "OBJECT",
                  properties: {
                    type: { type: "STRING" },
                    text: { type: "STRING" },
                    hashtags: { type: "STRING" }
                  },
                  required: ["type", "text", "hashtags"]
                }
              }
            },
            required: ["wishes"]
          }
        }
      };

      const targetUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
      const response = await fetchWithRetry(targetUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      const responseText = response.candidates?.[0]?.content?.parts?.[0]?.text;
      if (responseText) {
        const parsed = JSON.parse(responseText);
        if (parsed?.wishes) {
          setAiGeneratedWishes(parsed.wishes);
        }
      } else {
        throw new Error('Failed to parse structured wishes JSON.');
      }
    } catch (err) {
      console.error(err);
      setErrorMessage(`Failed to compose wishes: ${err.message}`);
    } finally {
      setIsGeneratingWishes(false);
    }
  };

  const generatePoster = async () => {
    if (!apiKey) {
      setErrorMessage('A valid Gemini API Key is required to run poster generation.');
      setApiInputVisible(true);
      return;
    }
    setIsGenerating(true);
    setErrorMessage('');
    setGeneratedImageUrl(null);

    // Prompt compilation based on UI settings
    let builtPrompt = '';
    
    if (composition === 'double-portrait') {
      builtPrompt += `Double portrait composition birthday poster. One large faded artistic portrait background with paint splash and ink spill effects. On the right side, a sharp full-body or close-up action pose. `;
    } else if (composition === 'flying-hair') {
      builtPrompt += `A dynamic studio fashion portrait. Trendy pose with dramatic flying wind-blown hair, highly stylized artistic layout. `;
    } else {
      builtPrompt += `A modern multi-exposure layered collage poster layout. Integrated design elements. `;
    }

    builtPrompt += `Subject: A highly stylish, attractive individual with the face identical to the source photo. Smooth skin, flawless facial details, 100% face likeness of the person. `;
    builtPrompt += `Styling & Fashion: Trendy high-fashion designer outfit. `;

    if (theme === 'gold-black') {
      builtPrompt += `Color scheme is luxury black, dark slate, and rich metallic gold accent splashes. Premium grand celebration atmosphere. `;
    } else if (theme === 'watercolor-ink') {
      builtPrompt += `Color scheme is artistic clean white and textured watercolor with colorful elegant ink splashes. Creative modern fluid layout. `;
    } else if (theme === 'cinematic-neon') {
      builtPrompt += `Color scheme is deep cyberpunk blues, hot pink highlights, and reflective glowing neon light traces. High-contrast futuristic vibe. `;
    } else {
      builtPrompt += `Color scheme is pure minimal white, subtle light gray drop shadows, clean borders, high-end editorial look. `;
    }

    if (lighting === 'cinematic-golden') {
      builtPrompt += `Lighting: Moody, dramatic side lighting, rich warm golden hour cinematic glow, beautiful catchlights in eyes. `;
    } else if (lighting === 'studio-soft') {
      builtPrompt += `Lighting: High-end professional multi-point softbox studio lighting, clean soft shadows, bright and crisp details. `;
    } else {
      builtPrompt += `Lighting: Atmospheric neon rims, backlighting, volumetric light beams cutting through particles. `;
    }

    builtPrompt += `Typography integrated directly in the artwork: elegant, professionally placed modern text banner reads exactly "${headline} ${targetName}". Bold design, premium layout. `;
    builtPrompt += `Overall style: 16k resolution, photorealistic, premium graphic poster design, commercial magazine cover grade. `;
    
    if (extraPrompt) {
      builtPrompt += ` Additional details: ${extraPrompt}`;
    }

    try {
      // Step 1: Pre-checking inputs
      setGenerationSteps('Analyzing and preparing layout templates...');
      await new Promise(r => setTimeout(r, 800));

      let resultUrl = '';

      if (imageSrc) {
        // Image to Image Flow (Gemini 2.5 Flash Image Preview)
        setGenerationSteps('Processing portrait & matching facial features with prompt...');
        
        // Extract raw base64 data from the dataURL
        const base64Data = imageSrc.split(',')[1];
        const mimeType = imageSrc.split(';')[0].split(':')[1] || 'image/jpeg';

        const payload = {
          contents: [{
            parts: [
              { text: `Redesign this uploaded image into a magnificent birthday celebration collage poster. Use the following detailed design recipe: ${builtPrompt}. Preserve the person's precise facial features, expressions, and likeness completely. Do not change their face. Output only the finished, high-fidelity generated image.` },
              { inlineData: { mimeType: mimeType, data: base64Data } }
            ]
          }],
          generationConfig: {
            responseModalities: ["IMAGE"]
          }
        };

        const targetUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-image-preview:generateContent?key=${apiKey}`;
        
        setGenerationSteps('Synthesizing double-portrait layers & ink-splash effects...');
        const result = await fetchWithRetry(targetUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });

        const base64ImageOut = result.candidates?.[0]?.content?.parts?.find(p => p.inlineData)?.inlineData?.data;
        if (!base64ImageOut) {
          throw new Error('Could not extract image from the model response. Please verify your API key or try again.');
        }

        resultUrl = `data:image/png;base64,${base64ImageOut}`;

      } else {
        // Text to Image Flow (Imagen 4.0 Generate)
        setGenerationSteps('Initializing text-to-image engine (Imagen 4.0)...');
        await new Promise(r => setTimeout(r, 600));
        setGenerationSteps('Rendering high-fidelity cinematic composition...');

        const payload = {
          instances: [
            { prompt: builtPrompt }
          ],
          parameters: {
            sampleCount: 1,
            aspectRatio: "2:3" // Perfect for mobile portrait wallpapers/posters
          }
        };

        const targetUrl = `https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${apiKey}`;

        const result = await fetchWithRetry(targetUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });

        const base64ImageOut = result.predictions?.[0]?.bytesBase64Encoded;
        if (!base64ImageOut) {
          throw new Error('Could not generate image. Please ensure your prompt or parameters are correct.');
        }

        resultUrl = `data:image/png;base64,${base64ImageOut}`;
      }

      setGenerationSteps('Polishing typography and colors...');
      await new Promise(r => setTimeout(r, 1000));

      setGeneratedImageUrl(resultUrl);
      
      // Save to local history
      const newHistoryItem = {
        id: Date.now().toString(),
        name: targetName,
        headline: headline,
        theme: theme,
        url: resultUrl,
        timestamp: new Date().toLocaleDateString()
      };
      const updatedHistory = [newHistoryItem, ...recentPosters].slice(0, 10);
      setRecentPosters(updatedHistory);
      localStorage.setItem('gemini_poster_history', JSON.stringify(updatedHistory));

    } catch (error) {
      console.error(error);
      setErrorMessage(error.message || 'An unexpected error occurred during generation.');
    } finally {
      setIsGenerating(false);
      setGenerationSteps('');
    }
  };

  const downloadGeneratedPoster = () => {
    if (!generatedImageUrl) return;
    const link = document.createElement('a');
    link.href = generatedImageUrl;
    link.download = `${targetName.replace(/\s+/g, '_')}_Birthday_Poster.png`;
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  const useHistoryItem = (item) => {
    setTargetName(item.name);
    setHeadline(item.headline);
    setTheme(item.theme);
    setGeneratedImageUrl(item.url);
  };

  const clearHistory = () => {
    setRecentPosters([]);
    localStorage.removeItem('gemini_poster_history');
  };

  return (
    <div className="min-h-screen bg-neutral-950 text-neutral-100 font-sans selection:bg-amber-500 selection:text-neutral-950">
      
      {/* Premium Header */}
      <header className="border-b border-neutral-800 bg-neutral-900/60 backdrop-blur-md sticky top-0 z-40 px-4 py-3">
        <div className="max-w-7xl mx-auto flex flex-col sm:flex-row items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-gradient-to-tr from-amber-500 via-rose-500 to-indigo-600 flex items-center justify-center shadow-lg shadow-amber-500/10">
              <SparklesIcon />
            </div>
            <div>
              <h1 className="text-xl font-extrabold tracking-tight bg-gradient-to-r from-amber-400 via-rose-400 to-indigo-400 bg-clip-text text-transparent">
                PosterStudio AI
              </h1>
              <p className="text-xs text-neutral-400">Cinematic Collage & Portrait Builder</p>
            </div>
          </div>

          <div className="flex items-center gap-2 w-full sm:w-auto">
            <button
              onClick={() => setApiInputVisible(!apiInputVisible)}
              className="px-4 py-2 rounded-lg bg-neutral-800 hover:bg-neutral-700 text-xs font-semibold flex items-center gap-2 transition ml-auto"
            >
              <SettingsIcon />
              {apiKey ? "API Key Configured" : "Set Gemini API Key"}
            </button>
          </div>
        </div>
      </header>

      {/* Main Studio Body */}
      <main className="max-w-7xl mx-auto p-4 lg:py-8 grid grid-cols-1 lg:grid-cols-12 gap-8">
        
        {/* Left Control Panel Column (5 cols) */}
        <section className="lg:col-span-5 flex flex-col gap-6">
          
          {/* API Key Banner/Input if not set */}
          {apiInputVisible && (
            <div className="p-4 rounded-xl bg-amber-500/10 border border-amber-500/30 text-amber-200">
              <div className="flex items-start gap-3 mb-2">
                <InfoIcon />
                <div className="text-xs">
                  <span className="font-bold">Required Setup:</span> Input your Gemini API Key below to utilize high-fidelity poster generation and AI assistant features. The key remains entirely stored on your browser.
                </div>
              </div>
              <input
                type="password"
                placeholder="AIzaSy..."
                value={apiKey}
                onChange={handleApiKeyChange}
                className="w-full mt-2 px-3 py-2 bg-neutral-900 border border-neutral-700 rounded-lg text-sm text-neutral-200 focus:outline-none focus:ring-1 focus:ring-amber-500"
              />
            </div>
          )}

          {/* Step 1: Upload Source Image */}
          <div className="p-5 rounded-2xl bg-neutral-900 border border-neutral-800 flex flex-col gap-4 shadow-xl">
            <div className="flex justify-between items-center">
              <h2 className="font-bold text-sm tracking-wider uppercase text-neutral-400">1. Upload Portrait</h2>
              {imageSrc && (
                <button 
                  onClick={removeUploadedImage} 
                  className="p-1 hover:bg-neutral-800 rounded transition text-xs flex items-center gap-1"
                >
                  <TrashIcon /> Remove
                </button>
              )}
            </div>

            <input
              type="file"
              accept="image/*"
              ref={fileInputRef}
              onChange={handleImageUpload}
              className="hidden"
            />

            {!imageSrc ? (
              <div 
                onClick={triggerFileUpload}
                className="border-2 border-dashed border-neutral-800 hover:border-amber-500/50 rounded-xl p-8 text-center cursor-pointer bg-neutral-950 hover:bg-neutral-900/40 transition duration-300 flex flex-col items-center justify-center gap-3"
              >
                <UploadIcon />
                <div>
                  <p className="text-sm font-semibold text-neutral-300">Choose portrait photo</p>
                  <p className="text-xs text-neutral-500 mt-1">PNG, JPG up to 10MB</p>
                </div>
              </div>
            ) : (
              <div className="relative rounded-xl overflow-hidden aspect-[4/5] bg-neutral-950 border border-neutral-800">
                <img 
                  src={imageSrc} 
                  alt="Uploaded Source Portrait" 
                  className="w-full h-full object-cover" 
                />
                <div className="absolute inset-x-0 bottom-0 bg-gradient-to-t from-neutral-950/90 to-transparent p-3 flex justify-between items-center">
                  <span className="text-xs text-emerald-400 flex items-center gap-1">
                    <span className="w-2 h-2 rounded-full bg-emerald-500 animate-pulse"></span>
                    Portrait Ready
                  </span>
                  <button 
                    onClick={triggerFileUpload} 
                    className="text-xs text-neutral-300 hover:text-white bg-neutral-900/80 px-2 py-1 rounded border border-neutral-700"
                  >
                    Change
                  </button>
                </div>
              </div>
            )}
          </div>

          {/* Step 2: Poster Content & Text Layering */}
          <div className="p-5 rounded-2xl bg-neutral-900 border border-neutral-800 flex flex-col gap-4 shadow-xl">
            <h2 className="font-bold text-sm tracking-wider uppercase text-neutral-400">2. Customize Typography Banner</h2>
            
            <div className="grid grid-cols-2 gap-3">
              <div>
                <label className="text-xs text-neutral-400 block mb-1">Occasion / Header</label>
                <input
                  type="text"
                  value={headline}
                  onChange={(e) => setHeadline(e.target.value)}
                  className="w-full px-3 py-2 bg-neutral-950 border border-neutral-800 focus:border-amber-500 rounded-lg text-sm text-neutral-200 outline-none"
                  placeholder="Happy Birthday"
                />
              </div>
              <div>
                <label className="text-xs text-neutral-400 block mb-1">Celebrant's Name</label>
                <input
                  type="text"
                  value={targetName}
                  onChange={(e) => setTargetName(e.target.value)}
                  className="w-full px-3 py-2 bg-neutral-950 border border-neutral-800 focus:border-amber-500 rounded-lg text-sm text-neutral-200 outline-none"
                  placeholder="Ibran Khan"
                />
              </div>
            </div>
          </div>

          {/* Step 3: Aesthetic Style Theme Options */}
          <div className="p-5 rounded-2xl bg-neutral-900 border border-neutral-800 flex flex-col gap-4 shadow-xl">

            <div>
              <label className="text-xs text-neutral-400 block mb-2">Composition Layout</label>

                <button
                  type="button"
                  onClick={() => setComposition('double-portrait')}
                  className={`py-2 px-1 rounded-lg border font-medium transition text-center ${composition === 'double-portrait' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}
                >
                  Double Portrait
                </button>
                <button
                  type="button"
                  onClick={() => setComposition('flying-hair')}
                  className={`py-2 px-1 rounded-lg border font-medium transition text-center ${composition === 'flying-hair' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}
                >

                </button>
                <button
                  type="button"
                  onClick={() => setComposition('poster-collage')}
                  className={`py-2 px-1 rounded-lg border font-medium transition text-center ${composition === 'poster-collage' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}
                >
                  Multi-Collage
                </button>
              </div>
            </div>

            {/* Theme Preset */}
            <div>
              <label className="text-xs text-neutral-400 block mb-2">Color Atmosphere Preset</label>
              <div className="grid grid-cols-2 gap-2 text-xs">
                <button
                  type="button"
                  onClick={() => setTheme('gold-black')}
                  className={`py-2.5 px-3 rounded-lg border font-medium transition text-left flex items-center justify-between ${theme === 'gold-black' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}
                >
                  <span>Luxury Gold & Black</span>
                  <span className="w-3 h-3 rounded-full bg-gradient-to-r from-yellow-500 to-black border border-neutral-700"></span>
                </button>
                <button
                  type="button"
                  onClick={() => setTheme('watercolor-ink')}
                  className={`py-2.5 px-3 rounded-lg border font-medium transition text-left flex items-center justify-between ${theme === 'watercolor-ink' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}
                >
                  <span>Ink Splash Paint</span>
                  <span className="w-3 h-3 rounded-full bg-gradient-to-r from-teal-400 to-rose-400 border border-neutral-700"></span>
                </button>
                <button
                  type="button"
                  onClick={() => setTheme('cinematic-neon')}
                  className={`py-2.5 px-3 rounded-lg border font-medium transition text-left flex items-center justify-between ${theme === 'cinematic-neon' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}
                >
                  <span>Cinematic Cyber Neon</span>
                  <span className="w-3 h-3 rounded-full bg-gradient-to-r from-pink-500 to-blue-500 border border-neutral-700"></span>
                </button>
                <button
                  type="button"
                  onClick={() => setTheme('minimalist-white')}
                  className={`py-2.5 px-3 rounded-lg border font-medium transition text-left flex items-center justify-between ${theme === 'minimalist-white' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}
                >
                  <span>Minimalist Clean White</span>
                  <span className="w-3 h-3 rounded-full bg-white border border-neutral-300"></span>
                </button>
              </div>
            </div>

            {/* Cinematic Lighting Preset */}
            <div>
              <label className="text-xs text-neutral-400 block mb-2">Cinematic Lighting</label>
              <div className="grid grid-cols-3 gap-2 text-xs">
                <button
                  type="button"
                  onClick={() => setLighting('cinematic-golden')}
                  className={`py-2 px-1 rounded-lg border font-medium transition text-center ${lighting === 'cinematic-golden' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}
                >
                  Golden Hour
                </button>
                <button
                  type="button"
                  onClick={() => setLighting('studio-soft')}
                  className={`py-2 px-1 rounded-lg border font-medium transition text-center ${lighting === 'studio-soft' ? 'bg-amber-500/10 border-amber-500 text-amber-400' : 'bg-neutral-950 border-neutral-800 text-neutral-400 hover:border-neutral-700'}`}

                  type="button"

                >
                  Neon Rim light
                </button>
              </div>
            </div>

            {/* Custom Prompt Box */}

              <div className="flex justify-between items-center mb-1">

                {/* ✨ LLM Prompt Enhancer Button */}
                <button
                  onClick={handleEnhancePrompt}
                  disabled={isEnhancingPrompt}

                  {isEnhancingPrompt ? <RefreshIcon /> : '✨ Expand with AI'}
                </button>
              </div>
              <textarea
                value={extraPrompt}
                onChange={(e) => setExtraPrompt(e.target.value)}
                className="w-full h-24 px-3 py-2 bg-neutral-950 border border-neutral-800 focus:border-amber-500 rounded-lg text-xs text-neutral-300 outline-none resize-none"
              />
              {llmNotification && (
                <p className="text-[10px] text-emerald-400 mt-1 font-medium">{llmNotification}</p>
              )}
            </div>
          </div>

          {/* Trigger Generate Button */}

            onClick={generatePoster}
            disabled={isGenerating}
            className={`w-full py-4 px-6 rounded-2xl font-bold flex items-center justify-center gap-2 transition duration-300 shadow-lg shadow-amber-500/10 ${

                ? 'bg-neutral-800 text-neutral-500 cursor-not-allowed'

          >
            {isGenerating ? (

                <RefreshIcon />
                <span>Running Generation...</span>

              <>
                <SparklesIcon />

              </>
            )}
          </button>

          {/* Error Message display */}
          {errorMessage && (
            <div className="p-4 rounded-xl bg-rose-500/10 border border-rose-500/30 text-rose-300 text-xs">
              <span className="font-semibold block mb-1">Could not process poster request:</span>
              {errorMessage}
            </div>
          )}

        </section>

        {/* Right Preview/Output Canvas (7 cols) */}
        <section className="lg:col-span-7 flex flex-col gap-6">
          
          <div className="rounded-2xl bg-neutral-900 border border-neutral-800 p-6 flex flex-col gap-4 shadow-xl flex-grow min-h-[500px]">
            
            <div className="flex justify-between items-center pb-2 border-b border-neutral-800">
              <div className="flex items-center gap-2">
                <span className="w-2.5 h-2.5 rounded-full bg-amber-500 animate-pulse"></span>
                <span className="text-sm font-semibold tracking-wider text-neutral-300">Live Preview & Output</span>
              </div>
              
              {generatedImageUrl && (
                <button
                  onClick={downloadGeneratedPoster}
                  className="px-3 py-1.5 bg-amber-500 hover:bg-amber-600 text-neutral-950 rounded-lg text-xs font-bold flex items-center gap-1.5 transition"
                >
                  <DownloadIcon />
                  Download Poster
                </button>
              )}
            </div>

            {/* Poster Canvas container */}
            <div className="flex-grow flex items-center justify-center relative w-full overflow-hidden bg-neutral-950 rounded-xl min-h-[400px]">
              
              {isGenerating ? (
                // Generating/Loading Screen
                <div className="absolute inset-0 bg-neutral-950/90 z-20 flex flex-col items-center justify-center p-6 text-center">
                  <div className="relative w-24 h-24 mb-6">
                    <div className="absolute inset-0 rounded-full border-4 border-dashed border-amber-500/30 animate-spin" style={{ animationDuration: '8s' }}></div>
                    <div className="absolute inset-2 rounded-full border-4 border-dashed border-rose-500/50 animate-spin" style={{ animationDuration: '4s', animationDirection: 'reverse' }}></div>
                    <div className="absolute inset-4 rounded-full border-4 border-neutral-800 flex items-center justify-center">
                      <SparklesIcon />
                    </div>
                  </div>
                  <h3 className="text-lg font-bold bg-gradient-to-r from-amber-400 to-rose-400 bg-clip-text text-transparent mb-1">
                    AI Artist Active
                  </h3>
                  <p className="text-xs text-neutral-400 max-w-sm">
                    {generationSteps || 'Compiling elements & layouts...'}
                  </p>
                </div>
              ) : null}

              {/* Poster Output Image or Layout preview placeholder */}
              {generatedImageUrl ? (
                <img 
                  src={generatedImageUrl} 
                  alt="AI Generated Celebration Poster" 
                  className="w-full max-w-lg aspect-[2/3] object-contain rounded-lg shadow-2xl border border-neutral-800"
                />
              ) : (
                // Placeholder mockup UI structure of the collage
                <div className="w-full max-w-sm aspect-[2/3] rounded-lg border-2 border-dashed border-neutral-800/80 p-6 flex flex-col justify-between relative bg-neutral-950 shadow-inner overflow-hidden">
                  
                  {/* Background faded aesthetic preview */}
                  <div className="absolute inset-0 bg-gradient-to-b from-neutral-950 via-neutral-900/40 to-neutral-950 z-0"></div>
                  
                  {/* Ink splash background placeholder */}
                  {theme === 'gold-black' && <div className="absolute top-1/4 left-1/4 w-40 h-40 rounded-full bg-yellow-600/10 filter blur-3xl"></div>}
                  {theme === 'watercolor-ink' && <div className="absolute top-1/4 left-1/4 w-40 h-40 rounded-full bg-teal-500/10 filter blur-3xl"></div>}
                  {theme === 'cinematic-neon' && <div className="absolute top-1/4 left-1/4 w-40 h-40 rounded-full bg-pink-600/15 filter blur-3xl"></div>}

                  {/* Header text placeholder mockup */}
                  <div className="relative z-10 text-center select-none pt-4">
                    <span className="text-[10px] tracking-[0.3em] uppercase text-neutral-400 font-bold block mb-1">CELEBRATION ANNOUNCEMENT</span>
                    <h2 className="text-2xl font-black uppercase text-white tracking-tight drop-shadow-md">
                      {headline || "HAPPY BIRTHDAY"}
                    </h2>
                  </div>

                  {/* Center Graphic composition placeholder representation */}
                  <div className="relative z-10 flex items-center justify-center flex-grow py-6">
                    {imageSrc ? (
                      <div className="relative w-40 h-40 flex items-center justify-center">
                        <img 
                          src={imageSrc} 
                          className="w-28 h-28 object-cover rounded-full border-4 border-neutral-800 absolute z-10 opacity-60 filter grayscale brightness-75 scale-90 -translate-x-6 -translate-y-4" 
                        />
                        <img 
                          src={imageSrc} 
                          className="w-28 h-28 object-cover rounded-full border-4 border-amber-500 absolute z-20" 
                        />
                        <div className="absolute bottom-[-10px] right-[-10px] bg-neutral-900 px-2 py-1 border border-neutral-700 rounded text-[9px] text-amber-400 font-semibold uppercase">
                          Double Portrait
                        </div>
                      </div>
                    ) : (
                      <div className="text-center text-neutral-600 max-w-xs px-4 select-none">
                        <div className="text-3xl mb-3">🎨</div>
                        <p className="text-xs font-semibold">Your Portrait & Style Collage Visualized Here</p>
                        <p className="text-[10px] text-neutral-700 mt-1">Upload an image to unlock high-fidelity Image-to-Image composition, or click "Generate" directly for text-only rendering.</p>
                      </div>
                    )}
                  </div>

                  {/* Footer Name Banner placeholder mockup */}
                  <div className="relative z-10 text-center select-none pb-4">
                    <p className="text-xs font-bold tracking-[0.2em] text-neutral-500 uppercase">HONORING</p>
                    <h3 className="text-3xl font-extrabold bg-gradient-to-r from-amber-400 via-rose-400 to-indigo-400 bg-clip-text text-transparent tracking-wide drop-shadow-lg mt-1">
                      {targetName || "IBRAN KHAN"}
                    </h3>
                  </div>

                </div>
              )}

            </div>
          </div>

          {/* ✨ INTEGRATION 3: Social Copywriting Companion Section */}
          <div className="p-5 rounded-2xl bg-neutral-900 border border-neutral-800 shadow-xl flex flex-col gap-4">
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
              <div>
                <h3 className="font-bold text-sm tracking-wider uppercase text-neutral-400">✨ AI Social Copywriting Companion</h3>
                <p className="text-xs text-neutral-500 mt-0.5">Generate stunning matching wishes, poetic shayari, and Instagram captions with hashtags.</p>
              </div>
              <button
                onClick={handleComposeWishes}
                disabled={isGeneratingWishes}
                className="px-3 py-2 bg-gradient-to-r from-indigo-600 to-rose-600 hover:opacity-90 disabled:bg-neutral-800 text-xs text-white rounded-lg font-bold flex items-center justify-center gap-1.5 transition self-start sm:self-auto"
              >
                {isGeneratingWishes ? <RefreshIcon /> : '✨ Compose Birthday Wishes'}
              </button>
            </div>

            {aiGeneratedWishes.length > 0 ? (
              <div className="grid grid-cols-1 md:grid-cols-3 gap-3 mt-1">
                {aiGeneratedWishes.map((wish, index) => (
                  <div key={index} className="p-4 rounded-xl bg-neutral-950 border border-neutral-800 flex flex-col justify-between relative group hover:border-indigo-500/40 transition">

                      <span className="text-[9px] uppercase font-bold tracking-widest text-indigo-400 block mb-2">{wish.type}</span>
                      <p className="text-xs text-neutral-200 italic leading-relaxed">"{wish.text}"</p>

                    </div>
                    <button
                      onClick={() => copyTextToClipboard(`${wish.text}\n\n${wish.hashtags}`, index)}

                      title="Copy to Clipboard"
                    >
                      {copySuccessId === index ? (
                        <span className="text-[9px] text-emerald-400 font-bold">Copied!</span>
                      ) : (
                        <CopyIcon />
                      )}
                    </button>
                  </div>
                ))}
              </div>
            ) : (
              <div className="p-8 border border-dashed border-neutral-800/80 rounded-xl text-center bg-neutral-950/40 select-none">
                <p className="text-xs text-neutral-500">No wishes generated yet. Click "✨ Compose Birthday Wishes" to write tailored, highly customized greetings matching your theme style!</p>
              </div>
            )}
          </div>

          {/* History / Gallery of recently made posters */}
          {recentPosters.length > 0 && (
            <div className="p-5 rounded-2xl bg-neutral-900 border border-neutral-800 shadow-xl">
              <div className="flex justify-between items-center mb-4">
                <h3 className="font-bold text-sm tracking-wider uppercase text-neutral-400">Poster History</h3>
                <button 
                  onClick={clearHistory}
                  className="text-xs text-neutral-500 hover:text-red-400 transition"
                >
                  Clear All
                </button>
              </div>
              <div className="grid grid-cols-5 gap-3">
                {recentPosters.map((item) => (
                  <div 
                    key={item.id}
                    onClick={() => useHistoryItem(item)}
                    className="aspect-[2/3] rounded-lg overflow-hidden border border-neutral-800 hover:border-amber-500 cursor-pointer relative group transition"
                  >
                    <img src={item.url} alt={item.name} className="w-full h-full object-cover" />
                    <div className="absolute inset-0 bg-neutral-950/70 opacity-0 group-hover:opacity-100 transition flex flex-col justify-end p-2 text-[9px]">
                      <span className="font-bold text-white truncate">{item.name}</span>
                      <span className="text-neutral-400 truncate">{item.headline}</span>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}

        </section>

      </main>

      {/* Aesthetic Footer */}
      <footer className="border-t border-neutral-900 bg-neutral-950 py-8 px-4 text-center mt-12 text-xs text-neutral-500">
        <p>© 2026 PosterStudio AI. Powered by Gemini & Imagen Generation Systems.</p>
        <p className="mt-1">Ensuring 100% original likeness preservation via non-linear generative neural edits.</p>
      </footer>

    </div>
  );
